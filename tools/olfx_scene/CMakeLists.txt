# OLFX: CC0 living-room scene. Pico (OpenXR) backend always; Oculus Go (VrApi) backend when OLFX_VRAPI_DIR is set.
if(OLFX_G2_BUILD)
    # Pico G2 APK: Pico Native SDK (VRActivity) drives the frame loop; we only draw the scene.
    add_library(olfx_g2 SHARED olfx_g2.cpp)
    set_target_properties(olfx_g2 PROPERTIES CXX_STANDARD 17)
    target_compile_definitions(olfx_g2 PRIVATE OLFX_HOST_CLASS="org/olfx/g2/G2Host")
    target_link_libraries(olfx_g2 PRIVATE EGL GLESv3 android log)
elseif(OLFX_GO_BUILD)
    # Oculus Go APK: needs the Meta Mobile SDK VrApi headers + libvrapi.so (fetched by tools/go/fetch_vrapi.sh, not in git)
    add_library(olfx_go SHARED olfx_go.cpp)
    set_target_properties(olfx_go PROPERTIES CXX_STANDARD 17)
    target_include_directories(olfx_go PRIVATE ${OLFX_VRAPI_DIR}/Include)
    add_library(vrapi SHARED IMPORTED)
    set_target_properties(vrapi PROPERTIES IMPORTED_LOCATION ${OLFX_VRAPI_DIR}/Libs/Android/${ANDROID_ABI}/Release/libvrapi.so)
    target_link_libraries(olfx_go PRIVATE vrapi EGL GLESv3 android log)
else()
    add_library(olfx_scene SHARED olfx_scene.cpp)
    set_target_properties(olfx_scene PROPERTIES CXX_STANDARD 17)
    target_include_directories(olfx_scene PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../src/external/openxr_includes)
    target_compile_definitions(olfx_scene PRIVATE XR_NO_PROTOTYPES XR_USE_PLATFORM_ANDROID XR_USE_GRAPHICS_API_OPENGL_ES)
    target_link_libraries(olfx_scene PRIVATE EGL GLESv3 android log)
endif()
